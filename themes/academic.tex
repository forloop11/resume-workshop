
%==============================================================================
%  Resume theme: Academic
%  Latin Modern, the face of TeX's own documents, in black, with centered headings between rules.
%
%  Written to be read by applicant tracking systems as well as by people:
%  one column, in reading order; a standard face, with every character in
%  the PDF as text; plain bullets; and no icons, text boxes, or graphics
%  beyond a rule. Defines the same commands as every other theme, so the
%  generated interim/*.tex files build with it unchanged.
%
%  Compile with:  pdflatex -jobname=resume input/format.tex   (run twice for hyperrefs)
%            or:  tectonic -Z search-path=. input/format.tex
%  Portable: uses only standard TeX Live / Overleaf packages, no custom class.
%==============================================================================

\documentclass[letterpaper,10pt]{article}

% --- Font ------------------------------------------------------------------
% Latin Modern, a Computer Modern: the face of papers set in TeX.
% It ships with TeX Live (lm); without it the text is set in Computer
% Modern, which looks the same.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[T1]{fontenc}
  \usepackage[utf8]{inputenc}
  \IfFileExists{lmodern.sty}{\usepackage{lmodern}}{}
  % So that text copied out of the PDF, or read by a parser, has "fi" and
  % "fl" as letters, not as single ligature characters.
  \IfFileExists{glyphtounicode.tex}{\input{glyphtounicode}\pdfgentounicode=1}{}
\else
  % Tectonic (XeTeX): the same face as an OpenType font, without the "fi",
  % "fl", and "ff" ligatures, which a parser may not read as letters.
  \usepackage{fontspec}
  \setmainfont{lmroman10}[Extension=.otf, UprightFont=*-regular, BoldFont=*-bold,
    ItalicFont=*-italic, BoldItalicFont=*-bolditalic, Ligatures=NoCommon]
\fi
\IfFileExists{microtype.sty}{\usepackage{microtype}}{}

\newcommand{\role}[1]{%
  \textit{\textbf{#1}}\par\vspace{1pt}%
}

\newcommand{\position}[3]{%
  \vspace{\jobGap}%
  \begin{tabularx}{\textwidth}{@{}X r@{}}
    \textbf{\large #1}\hspace{4pt}\textit{\small #2} &
    {\small #3} \\
  \end{tabularx}%
  \vspace{1pt}%
}

\newcommand{\subrole}[2]{%
  \vspace{3pt}%
  \begin{tabularx}{\textwidth}{@{}X r@{}}
    \textit{\textbf{#1}} &
    \textit{\small #2} \\
  \end{tabularx}%
  \vspace{1pt}%
}

\newcommand{\entry}[2]{%
  \textbf{#1:} #2\par\vspace{2pt}%
}

\newcommand{\blurb}[1]{%

}

\newcommand{\stack}[1]{%
